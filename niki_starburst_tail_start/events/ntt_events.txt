namespace = ntt

# Schedule after ordinary game-start setup, so generated origin neighbours exist.
country_event = {
    id = ntt.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        is_multiplayer = no
        galaxy_shape = starburst
        NOT = { has_country_flag = ntt_attempted }
        days_passed < 2
    }
    immediate = {
        set_country_flag = ntt_attempted
        country_event = { id = ntt.2 days = 1 }
    }
}

country_event = {
    id = ntt.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        is_multiplayer = no
        galaxy_shape = starburst
        days_passed < 3
        NOT = { has_country_flag = ntt_processed }
    }
    immediate = {
        set_country_flag = ntt_processed
        set_country_flag = ntt_v012_attempted
        if = {
            limit = {
                exists = capital_scope
                is_nomadic = no
                # These origins depend on multiple independently placed empires.
                NOT = { has_origin = origin_hegemon }
                NOT = { has_origin = origin_common_ground }
                NOT = { has_origin = origin_imperial_vassal }
                NOT = { has_origin = origin_payback }
                NOT = { has_origin = origin_broken_shackles }
            }
            ntt_prepare = yes
        }
        else = { country_event = { id = ntt.12 } }
    }
}

# Optional console recovery for an early start already moved by v0.1.0/0.1.1.
# Never invoked automatically on load; never repeat on a v0.1.2 attempt.
country_event = {
    id = ntt.20
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        is_multiplayer = no
        galaxy_shape = starburst
        days_passed < 30
        has_country_flag = ntt_complete
        NOT = { has_country_flag = ntt_v012_attempted }
        NOT = { has_country_flag = ntt_pending_verification }
    }
    immediate = {
        set_country_flag = ntt_v012_attempted
        remove_country_flag = ntt_complete
        ntt_prepare = yes
    }
}
